### pytest-ფრაგმენტები (გამოგონილი მონაცემი; `tests/conftest.py`-ის `db`, `people`, `item`)

import pytest
from sqlalchemy.exc import IntegrityError
from app import domain as D
from app.workflow import ConflictError, apply


def _neg(db, item, people):
    apply(db, item, people["op"], D.QC_OPERATOR, "send", {"verdict": D.NEGATIVE, "qc_comment": "ხარვეზი"})


def test_b_reassign_accepts_non_operator(db, people, item):                       # B-01
    apply(db, item, people["qcm"], D.QC_MANAGER, "reassign", {"operator_id": str(people["sv"].id)})
    assert item.operator_id == people["sv"].id


def test_b_reassign_bad_id_crashes(db, people, item):                            # B-01, B-06
    with pytest.raises(ValueError):
        apply(db, item, people["qcm"], D.QC_MANAGER, "reassign", {"operator_id": "abc"})


def test_b_request_approval_skips_dispute(db, people, item):                     # B-02
    _neg(db, item, people)
    apply(db, item, people["op"], D.QC_OPERATOR, "request_verdict_change",
          {"new_verdict": D.RECOMMENDATION, "comment": "x"})
    apply(db, item, people["sv"], D.SURVEYOR, "field_respond", {"response": D.R_DISAGREE, "comment": "x"})
    apply(db, item, people["mgr"], D.FIELD_MANAGER, "manager_approve", {})
    assert item.state == D.AWAITING_QCM
    apply(db, item, people["qcm"], D.QC_MANAGER, "approve_verdict_change", {})
    assert item.state == D.AWAITING_FIELD and item.qcm_decision is None


def test_b_stale_request_overrides_direct_change(db, people, item):              # B-03
    _neg(db, item, people)
    apply(db, item, people["op"], D.QC_OPERATOR, "request_verdict_change",
          {"new_verdict": D.RECOMMENDATION, "comment": "x"})
    apply(db, item, people["qcm"], D.QC_MANAGER, "change_verdict", {"new_verdict": D.POSITIVE, "comment": "x"})
    apply(db, item, people["qcm"], D.QC_MANAGER, "approve_verdict_change", {})
    assert item.verdict == D.RECOMMENDATION and item.state == D.AWAITING_FIELD


def test_b_qc_error_survives_verdict_change(db, people, item):                   # B-04
    _neg(db, item, people)
    apply(db, item, people["sv"], D.SURVEYOR, "field_respond", {"response": D.R_DISAGREE, "comment": "x"})
    apply(db, item, people["mgr"], D.FIELD_MANAGER, "manager_approve", {})
    apply(db, item, people["qcm"], D.QC_MANAGER, "qcm_field_right", {"confirm_positive": "on"})
    apply(db, item, people["qcm"], D.QC_MANAGER, "change_verdict", {"new_verdict": D.NEGATIVE, "comment": "x"})
    assert item.state == D.AWAITING_FIELD and item.qc_error is True


def test_b_nonexistent_surveyor_crashes(db, people, item):                       # B-06
    with pytest.raises(IntegrityError):
        apply(db, item, people["op"], D.QC_OPERATOR, "save_review",
              {"verdict": D.NEGATIVE, "qc_comment": "x", "surveyors": ["999999"]})
        db.commit()


def test_b_assign_manager_accepts_non_manager(db, people, item):                 # B-09

    assert item.field_manager_id == people["op2"].id

    assert item.code_list == ["ZZZ-99"]


def test_b_comment_bumps_version(db, people, item):                              # B-12
    _neg(db, item, people)
    v = item.version
    apply(db, item, people["sv"], D.SURVEYOR, "comment", {"comment": "კითხვა"})

        apply(db, item, people["op"], D.QC_OPERATOR, "edit", {"qc_comment": "ახალი", "version": v})

   - მენეჯერების ჯამი > KPI — სწორია (§7: ოქმი ყველა მენეჯერს ეთვლება);
   - ფილტრები (პერიოდი, ზონა, ოპერატორი, სახეობა, ვერდიქტი, სიმძიმე) მუშაობს;
   - დაზუსტება „შემოწმებულში“ არ ითვლება;
   - Excel — 6 ფურცელი.
   - კოორდინატორის „ოპერატორები“ — უკვე S-14.
7. **მეილი:**
   - იქმნება მხოლოდ „✉“ მონიშვნით და მხოლოდ კონტროლის როლიდან (დაბრუნება, „არ დათანხმებულის“ გადაწყვეტილება, კომენტარი — OPEN_QUESTIONS #10);

8. **ძველი მონაცემი (`derive_state`, გამოგონილი მწკრივებით):** §9-ის წესები ემთხვევა — დადებითი → closed; M → corrected / acknowledged; „არ ვეთანხმები“ + Q > S + O → მენეჯერთან; Q > S, O-ს გარეშე → QC მენეჯერთან; „შესწორდა“, Q < S → QC-ის გადასამოწმებელი; მხოლოდ O → მენეჯერთან; „პროცესშია“ → pending. გაურკვეველი — Q-03.
